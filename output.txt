=== Код x86 ===
section .text
global _start
_start:
    ; variable x
    ; variable sum
    mov eax, 1
    mov x, eax
    mov eax, 0
    mov sum, eax
L1:
    mov eax, x
    cmp eax, 10
    setle al
    movzx eax, al
    mov t1, eax
    cmp t1, 0
    je L2
    mov eax, sum
    add eax, x
    mov t2, eax
    mov eax, t2
    mov sum, eax
    mov eax, x
    add eax, 1
    mov t3, eax
    mov eax, t3
    mov x, eax
    jmp L1
L2:
    ; print sum
    mov eax, 1
    xor ebx, ebx
    int 0x80
